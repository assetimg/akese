function full_rmdir($target) {
    if (is_dir($target)) {
        $files = glob($target . '*', GLOB_MARK); 
        foreach ($files as $file) {
            full_rmdir($file);
        }
        return @rmdir($target);
    } elseif (is_file($target)) {
        return @unlink($target);
    }
    return false;
}
