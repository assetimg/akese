<?php
session_start();

use ZipArchive;

// === CONFIG ===
$auth_user = "admin";
$auth_pass = "zeus123";
$log_file  = __DIR__ . '/access_log.txt';
$protected_dirs = ['secret', 'config']; // folder yang dilindungi (nama relatif)

function isProtectedPath($path) {
    global $protected_dirs;
    foreach ($protected_dirs as $protected) {
        if (strpos(realpath($path), DIRECTORY_SEPARATOR . $protected) !== false) {
            return true;
        }
    }
    return false;
}

// === LOGIN ===
if (!isset($_SESSION['logged_in'])) {
    if ($_SERVER['REQUEST_METHOD'] === 'POST' && $_POST['user'] === $auth_user && $_POST['pass'] === $auth_pass) {
        $_SESSION['logged_in'] = true;
        file_put_contents($log_file, "[LOGIN] " . date('Y-m-d H:i:s') . "\n", FILE_APPEND);
        header("Location: ?");
        exit;
    }
    echo <<<HTML
<html><head><title>Login</title>
<script src="https://cdn.tailwindcss.com"></script>
<link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/font-awesome/6.5.0/css/all.min.css">
</head>
<body class="bg-gray-900 text-white flex items-center justify-center h-screen">
<form method="POST" class="bg-gray-800 p-8 rounded shadow-lg w-96">
    <h2 class="text-3xl mb-6 text-center">Login</h2>
    <input name="user" placeholder="Username" class="w-full p-3 mb-4 rounded bg-gray-700 text-white" required>
    <input name="pass" type="password" placeholder="Password" class="w-full p-3 mb-6 rounded bg-gray-700 text-white" required>
    <button class="w-full bg-green-600 hover:bg-green-700 py-2 rounded">Login</button>
</form></body></html>
HTML;
    exit;
}

$dir = isset($_GET['path']) ? realpath($_GET['path']) : getcwd();
if (!$dir || !is_dir($dir)) $dir = getcwd();

// Proteksi folder
if (isProtectedPath($dir)) {
    echo "<h1 style='color: red; text-align: center;'>Access Denied to Protected Folder</h1>";
    exit;
}

chdir($dir);

function log_action($action, $target = '') {
    global $log_file;
    $log = "[$action] " . date('Y-m-d H:i:s') . " - " . $target . "\n";
    file_put_contents($log_file, $log, FILE_APPEND);
}

// Logout
if (isset($_POST['logout'])) {
    session_destroy();
    log_action("LOGOUT");
    header("Location: ?");
    exit;
}

// Tambah folder
if (isset($_POST['newfolder']) && trim($_POST['newfolder']) !== '') {
    $newfolder = basename(trim($_POST['newfolder'])); // sanitize
    $fullNewFolder = $dir . DIRECTORY_SEPARATOR . $newfolder;
    if (!is_dir($fullNewFolder) && !isProtectedPath($fullNewFolder)) {
        mkdir($fullNewFolder);
        log_action("CREATE FOLDER", $fullNewFolder);
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

// Tambah file kosong
if (isset($_POST['newfile']) && trim($_POST['newfile']) !== '') {
    $newfile = basename(trim($_POST['newfile'])); // sanitize
    $fullNewFile = $dir . DIRECTORY_SEPARATOR . $newfile;
    if (!file_exists($fullNewFile) && !isProtectedPath($fullNewFile)) {
        file_put_contents($fullNewFile, "");
        log_action("CREATE FILE", $fullNewFile);
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

// Upload file
if (isset($_FILES['upload']) && $_FILES['upload']['error'] === UPLOAD_ERR_OK) {
    $uploadName = basename($_FILES['upload']['name']); // sanitize
    $uploadTmp = $_FILES['upload']['tmp_name'];
    $uploadPath = $dir . DIRECTORY_SEPARATOR . $uploadName;
    if (!isProtectedPath($uploadPath) && move_uploaded_file($uploadTmp, $uploadPath)) {
        log_action("UPLOAD FILE", $uploadPath);
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

// Delete file/folder (multi)
if (isset($_POST['delete_selected'])) {
    if (!empty($_POST['selected'])) {
        foreach ($_POST['selected'] as $target) {
            $realTarget = realpath($target);
            if ($realTarget && file_exists($realTarget) && !isProtectedPath($realTarget)) {
                if (is_file($realTarget)) unlink($realTarget);
                elseif (is_dir($realTarget)) rmdir_recursive($realTarget);
                log_action("DELETE", $realTarget);
            }
        }
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

function rmdir_recursive($dir) {
    foreach (scandir($dir) as $item) {
        if ($item === '.' || $item === '..') continue;
        $path = $dir . DIRECTORY_SEPARATOR . $item;
        if (is_dir($path)) {
            rmdir_recursive($path);
        } else {
            unlink($path);
        }
    }
    rmdir($dir);
}

// Rename form submit
if (isset($_POST['rename_from'], $_POST['rename_to'])) {
    $from = realpath($_POST['rename_from']);
    $toName = basename(trim($_POST['rename_to']));
    $to = dirname($from) . DIRECTORY_SEPARATOR . $toName;
    if ($from && file_exists($from) && !file_exists($to) && trim($toName) !== '' && !isProtectedPath($from) && !isProtectedPath($to)) {
        rename($from, $to);
        log_action("RENAME", "$from -> $to");
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

// Edit form submit
if (isset($_POST['savefile'], $_POST['filename'])) {
    $filename = realpath($_POST['filename']);
    $content = $_POST['savefile'];
    if ($filename && file_exists($filename) && !isProtectedPath($filename)) {
        file_put_contents($filename, $content);
        log_action("EDIT FILE", $filename);
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

// ZIP multiple files/folders
if (isset($_POST['zip_selected']) && !empty($_POST['selected'])) {
    $zipname = $dir . DIRECTORY_SEPARATOR . 'archive_' . date('Ymd_His') . '.zip';
    $zip = new ZipArchive();
    if ($zip->open($zipname, ZipArchive::CREATE) === TRUE) {
        foreach ($_POST['selected'] as $item) {
            $realItem = realpath($item);
            if ($realItem && file_exists($realItem) && !isProtectedPath($realItem)) {
                addToZip($zip, $realItem, basename($realItem));
            }
        }
        $zip->close();
        log_action("ZIP CREATED", $zipname);
        header("Location: ?path=" . urlencode($dir) . "&download=" . urlencode($zipname));
        exit;
    }
}

function addToZip($zip, $path, $localPath) {
    if (is_dir($path)) {
        $zip->addEmptyDir($localPath);
        foreach (scandir($path) as $item) {
            if ($item === '.' || $item === '..') continue;
            addToZip($zip, $path . DIRECTORY_SEPARATOR . $item, $localPath . '/' . $item);
        }
    } else {
        $zip->addFile($path, $localPath);
    }
}

// Unzip file
if (isset($_POST['unzip_file_submit']) && isset($_POST['unzip_file']) && isset($_POST['unzip_target'])) {
    $zipFile = realpath($_POST['unzip_file']);
    $extractTo = $dir . DIRECTORY_SEPARATOR . trim($_POST['unzip_target']);
    if ($zipFile && is_file($zipFile) && is_dir($extractTo) && !isProtectedPath($zipFile) && !isProtectedPath($extractTo)) {
        $zip = new ZipArchive();
        if ($zip->open($zipFile) === TRUE) {
            $zip->extractTo($extractTo);
            $zip->close();
            log_action("UNZIP", $zipFile . " to " . $extractTo);
        }
    }
    header("Location: ?path=" . urlencode($dir));
    exit;
}

// Preview file
if (isset($_GET['preview'])) {
    $file = realpath($_GET['preview']);
    if ($file && is_file($file) && !isProtectedPath($file)) {
        $ext = strtolower(pathinfo($file, PATHINFO_EXTENSION));

        <script src='https://cdn.tailwindcss.com'></script>

        } elseif (in_array($ext, ['jpg', 'jpeg', 'png', 'gif', 'webp'])) {
            echo "<img src='$file' class='max-w-full rounded'>";

            echo "<embed src='$file' width='100%' height='800px' type='application/pdf'>";

}

// Fungsi breadcrumb
function breadcrumb($dir) {
    $parts = explode(DIRECTORY_SEPARATOR, trim($dir, DIRECTORY_SEPARATOR));
    $build = '';
    echo '<div class="text-sm text-gray-400 mb-4">Path: ';
    foreach ($parts as $index => $part) {
        $build .= DIRECTORY_SEPARATOR . $part;
        $fullpath = realpath($build);
        echo '<a href="?path=' . urlencode($fullpath) . '" class="text-green-400 hover:underline">' . htmlspecialchars($part) . '</a>';
        if ($index < count($parts) - 1) echo ' / ';
    }
    echo '</div>';
}

// Fungsi listFiles dengan checkbox, ikon, link preview, edit, rename, delete
function listFiles($dir, $search = '') {
    $files = array_diff(scandir($dir), ['.', '..']);
    echo '<form method="POST" onsubmit="return confirm(\'Delete selected files/folders?\')" id="multiForm">';
    echo '<table class="w-full text-sm bg-gray-800 rounded">
        <thead><tr class="text-left border-b border-gray-600">
            <th class="py-2 px-3"><input type="checkbox" id="selectAll" onclick="toggleAll(this)"></th>
            <th>Name</th><th>Size</th><th>Actions</th>
        </tr></thead><tbody>';
    foreach ($files as $file) {
        if ($search && stripos($file, $search) === false) continue;
        $path = $dir . DIRECTORY_SEPARATOR . $file;
        $isDir = is_dir($path);
        $size = $isDir ? 'DIR' : filesize($path) . ' B';
        $name = htmlspecialchars($file);
        $icon = $isDir ? '<i class="fas fa-folder text-yellow-400 mr-2"></i>' : '<i class="fas fa-file text-gray-400 mr-2"></i>';
        $link = $isDir ? "?path=" . urlencode(realpath($path)) : "?download=" . urlencode(realpath($path));
        $preview = (!$isDir && is_file($path)) ? "<a href='?preview=" . urlencode(realpath($path)) . "' class='text-blue-400 ml-2'>Preview</a>" : "";
        $delete = "?delete=" . urlencode(realpath($path)) . "&path=" . urlencode($dir);
        $rename = "?rename=" . urlencode(realpath($path));
        $edit = (!$isDir && is_file($path)) ? "?edit=" . urlencode(realpath($path)) : "";
        echo "<tr>
            <td class='py-2 px-3'><input type='checkbox' name='selected[]' value='" . htmlspecialchars(realpath($path)) . "'></td>
            <td class='py-2 px-3'>$icon<a href='$link' class='hover:underline'>$name</a></td>
            <td>$size</td>
            <td>
                $preview
                " . ($edit ? "<a href='$edit' class='text-green-400 ml-2'>Edit</a>" : "") . "
                <a href='$rename' class='text-yellow-400 ml-2'>Rename</a>
                <a href='$delete' class='text-red-400 ml-2' onclick=\"return confirm('Delete $name?')\">Delete</a>
            </td>
        </tr>";
    }
    echo '</tbody></table>';

    echo '<div class="mt-4 flex gap-4">';
    echo '<button type="submit" name="delete_selected" class="bg-red-600 px-4 py-2 rounded hover:bg-red-700">Delete Selected</button>';
    echo '<button type="submit" name="zip_selected" class="bg-purple-600 px-4 py-2 rounded hover:bg-purple-700">Create ZIP from Selected</button>';
    echo '</div>';

    echo '</form>';
    echo <<<JS
<script>
function toggleAll(source) {
  checkboxes = document.querySelectorAll('input[name="selected[]"]');
  for(let i=0; i < checkboxes.length; i++) {
    checkboxes[i].checked = source.checked;
  }
}
</script>
JS;
}
